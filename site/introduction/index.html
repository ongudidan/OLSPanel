Command line

OLSPanel Password reset

olspanel reset_admin_password my_new_password

Add new user

olspanel adduser --username john --first_name OLS --last_name Panel --email email@olspanel.com  --password securepassword --domain example2.com --pkg_id 1 --php_version 8.1

Issue SSL

olspanel ssl --domain example.com

Add new domain

olspanel domainadd --username john --domain example5.com --php_version 8.1 --path public_html

Suspend user

olspanel actionuser --username john --state suspend

UnSuspend user

olspanel actionuser --username john --state unsuspend

Delete Suspended user

olspanel actionuser --username john --state delete

Disable 2FA

olspanel actionuser --username john --state disable_otp

Edit user

olspanel edituser --username john --first_name John --last_name Doe --email john.doe@example.com --pkg_id 1

Change user password

olspanel edituser --username john --first_name John --last_name Doe --email john.doe@example.com --pkg_id 1 --password newsecurepassword

New Package

olspanel newpkg --name "Starter Plan" --disk_space 5000 --bandwidth 10000 --email_accounts 50 --databases 10 --ftp_accounts 5 --allowed_domains 2 --allowed_subdomains 5 --limit_action 0

New Database

olspanel makedb --username=babu --dbname=test --dbuser=testuser --dbpass=Pass1234 --dbpassc=Pass1234

New Cronjob

olspanel cronjob_add --username=babau --minute="0" --hour="3" --day="*" --month="*" --weekday="*" --comm="/home/babau/scripts/backup.sh"

Fix permission

olspanel fixperm --username=babu --full_path=/home/babu/public_html

Package List

olspanel pkglist

Make a Backup

olspanel backupnow --username john --backup_option full,database,email,dns,files --category default

OLSPanel Restart

systemctl restart cp

Openlitespeed Restart

/usr/local/lsws/bin/lswsctrl restart

MarialDB Restart

systemctl restart mariadb

Swap Memory add 2GB

curl -sSL https://olspanel.com/extra/swap.sh | sed 's/\r$//' | bash

Upgrade/Update

curl -sSL https://olspanel.com/upgrade.sh | sed 's/\r$//' | bash

Downgrade

VERSION=3.0.16 bash <(curl -sSL https://olspanel.com/downgrade.sh)
